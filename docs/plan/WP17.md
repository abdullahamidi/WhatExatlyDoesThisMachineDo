```ts
export interface GreedyOptions { /* existing */ snapshotAt?: readonly TechId[] }   // record state the step each id is first owned (also t = 0)
export interface GreedyResult { /* existing */ snapshots?: Partial<Record<TechId, GameState>> } // key present ONLY when snapshotAt given
export interface LedgerEntry { t: number; id: TechId; res: ResourceId; level: number /* new level */; cost: number }
export function purchaseLedger(purchases: GreedyResult['purchases'], techs?: Record<TechId, TechDefinition>,
  initialLevels?: Readonly<Record<TechId, number>> /* default createInitialState(0).techLevels */): LedgerEntry[]   // cost = costAt(def, levelBefore)
export function deadMinutes(ledger: readonly LedgerEntry[], res: ResourceId, endT: number): number   // (endT − last t ≤ endT paid in res)/60; none → endT/60
export function spendShare(ledger: readonly LedgerEntry[], res: ResourceId, fromT: number, toT: number,
  lifetimeFrom: number, lifetimeTo: number): number   // Σ cost (res, fromT < t ≤ toT) / (lifetimeTo − lifetimeFrom); produced ≤ 0 → 0
```
Snapshot taken after `record(t)` (after that step's purchases). Without `snapshotAt` the result is bit-identical to today (no `snapshots` key).
